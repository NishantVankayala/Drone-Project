<script>
  const API_BASE = window.location.protocol === "file:" ? "http://127.0.0.1:5000" : window.location.origin;
  let map;
  let markersLayer;
  let droneMarker = null;
  let latestAlerts = [];

  const droneIcon = L.divIcon({
    className: "drone-icon",
    html: `<div style="font-size:28px; filter: drop-shadow(0 0 5px #00aaff);">🚁</div>`,
    iconSize: [32, 32],
    iconAnchor: [16, 16]
  });

  function addLog(message) {
    const log = document.getElementById("log");
    const time = new Date().toLocaleTimeString();
    log.innerHTML = `[${time}] ${message}<br>` + log.innerHTML;
  }

  function initMap(lat = 17.385044, lon = 78.486671) {
    map = L.map("map").setView([lat, lon], 14);
    L.tileLayer("https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png", { maxZoom: 19, attribution: "© OpenStreetMap" }).addTo(map);
    markersLayer = L.layerGroup().addTo(map);
  }

  function statusClass(status) {
    if (status === "verified") return "verified";
    if (status === "false_alert") return "false-alert";
    if (status === "closed") return "closed";
    return "new";
  }

  function updateLiveTelemetry(t, count) {
    document.getElementById("alert-count").innerText = count;
    if (!t) return;

    document.getElementById("altitude").innerText = `${Number(t.altitude || 0).toFixed(2)} m`;
    document.getElementById("gps-ok").innerText = t.gps_ok ? "OK" : "WAIT";
    document.getElementById("confidence").innerText = Number(t.latest_confidence || 0).toFixed(3);
    document.getElementById("latest-label").innerText = t.latest_label || "--";
    document.getElementById("fire-status").innerText = t.fire_confirmed ? "CONFIRMED" : (t.fire_detected ? "CHECKING" : "CLEAR");
    document.getElementById("coords").innerHTML = `Drone Coordinates:<br>LAT: ${t.latitude}<br>LON: ${t.longitude}`;

    const status = document.getElementById("system-status");
    if (t.fire_confirmed) { status.className = "status danger"; status.innerText = "CONFIRMED FIRE ALERT"; }
    else if (t.fire_detected) { status.className = "status warn"; status.innerText = "CHECKING FIRE/SMOKE"; }
    else if (t.gps_ok) { status.className = "status"; status.innerText = "DRONE TELEMETRY ONLINE"; }
    else { status.className = "status warn"; status.innerText = "GPS WAIT"; }

    if (t.gps_ok && Number(t.latitude) !== 0 && Number(t.longitude) !== 0) {
      const pos = [Number(t.latitude), Number(t.longitude)];
      if (!droneMarker) {
        droneMarker = L.marker(pos, { icon: droneIcon }).addTo(map).bindPopup("Live Drone Position");
      } else {
        droneMarker.setLatLng(pos);
      }
    }
  }

  function updateLatestSnapshot(alerts) {
    const latestWithImage = alerts.find(a => a.image);
    if (latestWithImage) {
      document.getElementById("camera-content").className = "";
      document.getElementById("camera-content").innerHTML = `<img src="data:image/jpeg;base64,${latestWithImage.image}" alt="Latest alert snapshot">`;
    }
  }

  function updateTable(alerts) {
    const body = document.getElementById("alerts-body");
    if (!alerts || alerts.length === 0) {
      body.innerHTML = `<tr><td colspan="11">No alerts stored in SQLite.</td></tr>`;
      return;
    }

    body.innerHTML = alerts.map(alert => `
      <tr>
        <td>${alert.id}</td>
        <td>${alert.timestamp}</td>
        <td>${alert.drone_id}</td>
        <td>${alert.event_type}</td>
        <td>${alert.label}</td>
        <td>${alert.confidence}</td>
        <td>Lat: ${alert.latitude}<br>Lon: ${alert.longitude}</td>
        <td>${alert.altitude} m</td>
        <td>${alert.image ? `<img class="thumb" src="data:image/jpeg;base64,${alert.image}">` : "No image"}</td>
        <td class="${statusClass(alert.status)}">${alert.status}</td>
        <td>
          <button class="action-btn verify-btn" onclick="setAlertStatus(${alert.id}, 'verified')">Verify</button>
          <button class="action-btn false-btn" onclick="setAlertStatus(${alert.id}, 'false_alert')">False Alert</button>
          <button class="action-btn close-btn" onclick="setAlertStatus(${alert.id}, 'closed')">Close</button>
          <button class="action-btn delete-btn" onclick="deleteAlert(${alert.id})">Delete</button>
        </td>
      </tr>
    `).join("");
  }

  function updateMap(alerts) {
    markersLayer.clearLayers();
    if (!alerts || alerts.length === 0) return;

    alerts.forEach(alert => {
      if (!alert.latitude || !alert.longitude) return;
      let popup = `
        <b>Alert ID:</b> ${alert.id}<br>
        <b>Drone:</b> ${alert.drone_id}<br>
        <b>Event:</b> ${alert.event_type}<br>
        <b>Label:</b> ${alert.label}<br>
        <b>Confidence:</b> ${alert.confidence}<br>
        <b>Status:</b> ${alert.status}<br>
        <b>Altitude:</b> ${alert.altitude} m<br>
        <b>Time:</b> ${alert.timestamp}<br>
        <b>Lat:</b> ${alert.latitude}<br>
        <b>Lon:</b> ${alert.longitude}<br>`;
      if (alert.image) popup += `<img class="popup-img" src="data:image/jpeg;base64,${alert.image}">`;
      L.marker([alert.latitude, alert.longitude]).addTo(markersLayer).bindPopup(popup);
    });
  }

  async function setAlertStatus(alertId, status) {
    try {
      const response = await fetch(`${API_BASE}/api/alerts/${alertId}/status`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ status })
      });
      if (!response.ok) throw new Error("Status update failed");
      addLog(`[ACTION] Alert ${alertId} marked as ${status}`);
      await loadAll();
    } catch (error) { addLog("[ERROR] " + error.message); }
  }

  async function deleteAlert(alertId) {
    if (!confirm(`Delete alert ID ${alertId}?`)) return;
    try {
      const response = await fetch(`${API_BASE}/api/alerts/${alertId}`, { method: "DELETE" });
      if (!response.ok) throw new Error("Delete failed");
      addLog(`[ACTION] Alert ${alertId} deleted`);
      await loadAll();
    } catch (error) { addLog("[ERROR] " + error.message); }
  }

  async function deleteAllAlerts() {
    if (!confirm("Delete ALL alerts from SQLite? This cannot be undone.")) return;
    try {
      const response = await fetch(`${API_BASE}/api/alerts`, { method: "DELETE" });
      if (!response.ok) throw new Error("Delete all failed");
      addLog("[ACTION] All alerts deleted");
      await loadAll();
    } catch (error) { addLog("[ERROR] " + error.message); }
  }

  async function sendDroneCommand(command, params = {}) {
    if (command === "emergency_stop" && !confirm("Send EMERGENCY STOP command? Use only in SITL/testing unless safety verified.")) return;
    try {
      const response = await fetch(`${API_BASE}/api/drone-command`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ command, params })
      });
      const data = await response.json();
      if (!response.ok || !data.ok) throw new Error(data.message || "Command failed");
      addLog(`[COMMAND] ${command}: ${data.message}`);
    } catch (error) { addLog(`[ERROR] ${command}: ${error.message}`); }
  }

  async function loadAll() {
    try {
      const [alertsResponse, telemetryResponse] = await Promise.all([
        fetch(`${API_BASE}/api/alerts`),
        fetch(`${API_BASE}/api/telemetry`)
      ]);
      const alerts = await alertsResponse.json();
      const telemetry = await telemetryResponse.json();
      latestAlerts = alerts;
      updateLiveTelemetry(telemetry, alerts.length);
      updateLatestSnapshot(alerts);
      updateTable(alerts);
      updateMap(alerts);

      // Keep map centered on live drone when available, otherwise latest alert.
      if (telemetry.gps_ok && Number(telemetry.latitude) !== 0 && Number(telemetry.longitude) !== 0) {
        map.setView([Number(telemetry.latitude), Number(telemetry.longitude)], 14);
      } else if (alerts.length > 0) {
        map.setView([alerts[0].latitude, alerts[0].longitude], 14);
      }
    } catch (error) {
      addLog("[ERROR] Dashboard API not reachable: " + error.message);
      document.getElementById("system-status").className = "status danger";
      document.getElementById("system-status").innerText = "API OFFLINE";
    }
  }

  initMap();
  loadAll();
  setInterval(loadAll, 1000);
</script>
